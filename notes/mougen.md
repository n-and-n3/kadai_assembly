↓ コード例くれたありがとう
さっき言ったABI の例
(Linux の System V API に従って printf を call して`Hello World` を出力するアセンブリ)

```asm
.global main
.text

main:
    # スタックフレームの構築
    pushq   %rbp
    movq    %rsp, %rbp

    # 第1引数(書式付き文字列のポインタ) を %rdi に設定
    leaq    .LC0(%rip), %rdi

    # 浮動小数点引数の数 (printfの仕様で %al / %eax に指定; 今回は0個)
    movl    $0, %eax

    # printf 関数の呼び出し
    call    printf

    # 戻り値 0 を設定
    movl    $0, %eax

    # スタックフレームの破棄
    movq    %rbp, %rsp
    popq    %rbp
    ret

.section .rodata
.LC0:
    .string "Hello, world!\n"
```

計算機室の環境はintel版のLinuxらしい
